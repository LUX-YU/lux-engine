cmake_minimum_required(VERSION 3.28)
project(ObjectExecutionConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CONSUMER_MODE BRIDGE CACHE STRING "OBJECT or BRIDGE")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
if(CONSUMER_MODE STREQUAL "OBJECT")
    find_package(lux-engine-core REQUIRED COMPONENTS object)
    add_executable(consumer "${engine_source}/modules/core/object/test/target.cpp")
    target_link_libraries(consumer PRIVATE lux::engine::core::object)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME installed.object.target COMMAND consumer)
    add_test(NAME installed.object.target_contract COMMAND "${Python3_EXECUTABLE}"
        "${engine_source}/modules/core/object/test/target_contract.py" $<TARGET_FILE:consumer>)
    set(header object/ObjectTarget)
    set(provider lux::engine::core::object)
    if(TARGET lux::engine::process::process_execution OR TARGET STDEXEC::stdexec)
        message(FATAL_ERROR "Object-only consumer imported Process/stdexec")
    endif()
elseif(CONSUMER_MODE STREQUAL "BRIDGE")
    find_package(lux-engine-process-object-execution REQUIRED COMPONENTS object_execution)
    add_executable(consumer "${engine_source}/engine/process/object_execution/test/ObjectSchedulerTest.cpp")
    target_link_libraries(consumer PRIVATE lux::engine::process::object_execution)
    add_test(NAME installed.process.object_scheduler COMMAND consumer)
    set(header process/ObjectScheduler)
    set(provider lux::engine::process::object_execution)
else()
    message(FATAL_ERROR "Unknown CONSUMER_MODE")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/header.cpp" "#include <lux/engine/${header}.hpp>\n")
add_library(standalone_header OBJECT "${CMAKE_BINARY_DIR}/header.cpp")
target_link_libraries(standalone_header PRIVATE ${provider})
foreach(target consumer standalone_header)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /permissive- /GR- /we4541)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG -fno-rtti)
    endif()
endforeach()
if(TARGET lux::engine::editor::lux_editor_context OR TARGET lux::engine::function::ui)
    message(FATAL_ERROR "Object bridge imported Editor/UI")
endif()
