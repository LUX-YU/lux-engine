cmake_minimum_required(VERSION 3.22)
project(editor_sessions_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(lux-engine-edit-sessions CONFIG REQUIRED COMPONENTS edit_sessions)
add_executable(sessions_consumer ../../../editor/editing/test/sessions/sessions.cpp)
target_link_libraries(sessions_consumer PRIVATE lux::engine::editor::edit_sessions)
if(MSVC)
    target_compile_options(sessions_consumer PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(sessions_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.editor_sessions COMMAND sessions_consumer)
