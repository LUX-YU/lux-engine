cmake_minimum_required(VERSION 3.24)
project(inspector_generation_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-cxx REQUIRED COMPONENTS reflection_generator)
find_package(lux-engine-core REQUIRED COMPONENTS meta serialization)
find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS schema)
find_package(lux-engine-editor-scene-ui REQUIRED COMPONENTS scene_ui)
include_component_cmake_scripts(lux::engine::core::meta)
include_component_cmake_scripts(lux::engine::simulation::ecs::schema)
# The qualification driver mutates its own copy of the installed templates/support, never the live SDK.
include("${INSPECTOR_CMAKE_MODULE}")
set(LUX_GENERATE_HEADER_DIR "${CMAKE_BINARY_DIR}/generated")
add_executable(generation_consumer main.cpp)
set_property(TARGET generation_consumer PROPERTY LUX_ARCH_LAYER EDITOR)
target_link_libraries(generation_consumer PRIVATE lux::engine::editor::scene_ui)
target_include_directories(generation_consumer PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${LUX_GENERATE_HEADER_DIR}")
target_compile_definitions(generation_consumer PRIVATE "EC3_GENERATION_FLAG=${EC3_GENERATION_FLAG}")
if(MSVC)
    target_compile_options(generation_consumer PRIVATE /utf-8 /UNDEBUG /Zc:__cplusplus /Zc:preprocessor /permissive-)
else()
    target_compile_options(generation_consumer PRIVATE -UNDEBUG)
endif()
foreach(part a b)
    engine_target_add_ecs_component_codegen(NAME ${part}_schema TARGET generation_consumer
        SOURCE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" HEADER "${CMAKE_CURRENT_SOURCE_DIR}/include/${part}/Component.hpp"
        LOGICAL_PATH ${part}/Component.hpp SYMBOL ${part})
    engine_target_add_imgui_inspector_codegen(NAME ${part}_fields TARGET generation_consumer
        SOURCE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/main.cpp" HEADER "${CMAKE_CURRENT_SOURCE_DIR}/include/${part}/Component.hpp"
        LOGICAL_PATH ${part}/Component.hpp COMPONENTS ${part}::Component)
endforeach()
