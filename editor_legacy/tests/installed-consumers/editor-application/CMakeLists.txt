cmake_minimum_required(VERSION 3.24)
project(editor_application_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT SDK_PREFIX)
    message(FATAL_ERROR "SDK_PREFIX must select the installed SDK")
endif()
find_package(lux-engine-editor-application REQUIRED COMPONENTS editor_bootstrap)
find_package(lux-engine-editor-project-module REQUIRED COMPONENTS project_module)
add_executable(application_consumer main.cpp)
target_link_libraries(application_consumer PRIVATE lux::engine::editor::editor_bootstrap lux::engine::editor::project_module)
if(MSVC)
    target_compile_options(application_consumer PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
else()
    target_compile_options(application_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.application.command_thread
    COMMAND application_consumer "${SDK_PREFIX}" "${CMAKE_BINARY_DIR}/project")
set_tests_properties(installed.application.command_thread PROPERTIES LABELS "installed;P13;gpu" TIMEOUT 60)

find_package(lux-engine-editor-material-module REQUIRED COMPONENTS material_module)
foreach(forbidden flow_ui flow_module flowforge_persistence)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "Host without Flow imports ${forbidden}")
    endif()
endforeach()
add_executable(application_no_flow main.cpp)
target_compile_definitions(application_no_flow PRIVATE EC4_NO_FLOW=1)
target_link_libraries(application_no_flow PRIVATE lux::engine::editor::editor_bootstrap lux::engine::editor::project_module lux::engine::editor::material_module)
if(MSVC)
    target_compile_options(application_no_flow PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
else()
    target_compile_options(application_no_flow PRIVATE -UNDEBUG)
endif()
add_test(NAME installed.application.no_flow
    COMMAND application_no_flow "${SDK_PREFIX}" "${CMAKE_BINARY_DIR}/without-flow")
set_tests_properties(installed.application.no_flow PROPERTIES LABELS "installed;EC4;gpu" TIMEOUT 60)
