cmake_minimum_required(VERSION 3.24)
project(editor_component_closure LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Component qualification requires RelWithDebInfo")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

function(check_component package component header symbol)
    find_package(${package} REQUIRED COMPONENTS ${component})
    if(TARGET assimp::assimp)
        set_property(TARGET assimp::assimp PROPERTY MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
    endif()
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${component}.cpp")
    file(WRITE "${source}" "#include <${header}>\nvolatile auto entry = &${symbol};\nint main() { return entry == nullptr; }\n")
    add_executable(${component}_consumer "${source}")
    target_link_libraries(${component}_consumer PRIVATE lux::engine::editor::${component})
    target_compile_options(${component}_consumer PRIVATE /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    add_test(NAME closure.${component} COMMAND ${component}_consumer)
endfunction()
