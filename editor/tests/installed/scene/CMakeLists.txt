cmake_minimum_required(VERSION 3.28)
project(SceneFileConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-scene-asset REQUIRED COMPONENTS scene_asset)
add_executable(scene_consumer ../../../../engine/scene/asset/test/ScenePackageFile.cpp)
target_link_libraries(scene_consumer PRIVATE lux::engine::scene::scene_asset)
if(MSVC)
    target_compile_options(scene_consumer PRIVATE /UNDEBUG /utf-8 /GR- /we4541 /permissive-)
else()
    target_compile_options(scene_consumer PRIVATE -UNDEBUG -fno-rtti)
endif()
enable_testing()
add_test(NAME sdk.scene_file COMMAND scene_consumer "${CMAKE_CURRENT_BINARY_DIR}/data")
