cmake_minimum_required(VERSION 3.24)
project(editor_quality_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(lux-engine-editor-material-ui REQUIRED COMPONENTS material_ui)
find_package(lux-engine-editor-tasks-ui REQUIRED COMPONENTS tasks_ui)
find_package(lux-engine-editor-desktop REQUIRED COMPONENTS view_host)
find_package(lux-cxx REQUIRED COMPONENTS container)
foreach(forbidden scene_ui editor_ui editor_scene editor_material editor_flowforge editor_context editor_storage editor_app)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "Quality consumer imported unrelated or legacy target ${forbidden}")
    endif()
endforeach()
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
function(qualification name source)
    configure_file("${engine_source}/${source}" "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" COPYONLY)
    add_executable(${name} "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    target_link_libraries(${name} PRIVATE ${ARGN})
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME installed.quality.${name} COMMAND ${name})
endfunction()
