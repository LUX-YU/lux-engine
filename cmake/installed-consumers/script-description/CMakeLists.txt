cmake_minimum_required(VERSION 3.22)
project(lux_script_description_consumer LANGUAGES CXX)
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-engine-scene-script-description REQUIRED COMPONENTS scene_script_description)
get_target_property(description_links lux::engine::scene::scene_script_description INTERFACE_LINK_LIBRARIES)
if(description_links MATCHES "process|scene_script_runtime|scene_composition")
    message(FATAL_ERROR "Description leaf exports a Scene runtime/Process dependency: ${description_links}")
endif()
add_executable(lux_script_description_consumer main.cpp)
target_compile_features(lux_script_description_consumer PRIVATE cxx_std_20)
target_compile_options(lux_script_description_consumer PRIVATE /UNDEBUG)
target_link_libraries(lux_script_description_consumer PRIVATE lux::engine::scene::scene_script_description)
