cmake_minimum_required(VERSION 3.28)
project(ProjectManifestConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-framework REQUIRED COMPONENTS lux_editor_project)
add_executable(project_consumer ../../project_manifest.cpp)
target_link_libraries(project_consumer PRIVATE lux::engine::editor::lux_editor_project)
if(MSVC)
    target_compile_options(project_consumer PRIVATE /UNDEBUG /utf-8 /GR- /we4541 /permissive-)
else()
    target_compile_options(project_consumer PRIVATE -UNDEBUG -fno-rtti)
endif()
enable_testing()
add_test(NAME sdk.project_manifest COMMAND project_consumer "${CMAKE_CURRENT_BINARY_DIR}/data")
