cmake_minimum_required(VERSION 3.28)
project(EditorToolingConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Tooling consumer requires RelWithDebInfo")
endif()
find_package(lux-engine-editor-tooling REQUIRED COMPONENTS editor_tooling)
add_library(consumer_tool SHARED DLLTool.cpp)
target_compile_definitions(consumer_tool PRIVATE CONSUMER_TOOL_EXPORTS)
target_link_libraries(consumer_tool PUBLIC lux::engine::editor::editor_tooling)
add_executable(editor_tooling_consumer main.cpp)
target_link_libraries(editor_tooling_consumer PRIVATE consumer_tool)
target_compile_options(editor_tooling_consumer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
enable_testing()
add_test(NAME installed_editor_tooling COMMAND editor_tooling_consumer)
