cmake_minimum_required(VERSION 3.25)
project(p10q_before_bq1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-scene-model REQUIRED COMPONENTS scene_model)
add_executable(bq1 main.cpp)
target_include_directories(bq1 PRIVATE "${BASELINE_SOURCE}/editor/tools/scene/model/src")
target_link_libraries(bq1 PRIVATE lux::engine::editor::scene_model)
if(MSVC)
 target_compile_options(bq1 PRIVATE /UNDEBUG /utf-8 /permissive-)
endif()
