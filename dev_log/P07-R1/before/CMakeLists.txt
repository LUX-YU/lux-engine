cmake_minimum_required(VERSION 3.22)
project(p07_r1_before LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(lux-engine-editor-material-preview REQUIRED COMPONENTS material_preview)
find_package(lux-engine-editor-flowforge-compilation REQUIRED COMPONENTS flowforge_compilation)
add_executable(before before.cpp)
add_executable(traits traits.cpp)
add_executable(contract EXCLUDE_FROM_ALL contract.cpp)
foreach(t before traits contract)
 target_link_libraries(${t} PRIVATE lux::engine::editor::material_preview lux::engine::editor::flowforge_compilation)
 target_compile_options(${t} PRIVATE /UNDEBUG /utf-8)
endforeach()
