cmake_minimum_required(VERSION 3.22)
project(lux_editor_scene_consumer LANGUAGES CXX)
find_package(lux-engine-editor-scene REQUIRED COMPONENTS editor_scene)
add_executable(lux_editor_scene_consumer main.cpp)
target_compile_features(lux_editor_scene_consumer PRIVATE cxx_std_20)
target_link_libraries(lux_editor_scene_consumer PRIVATE lux::engine::editor::editor_scene)
if(MSVC)
    target_compile_options(lux_editor_scene_consumer PRIVATE /UNDEBUG)
else()
    target_compile_options(lux_editor_scene_consumer PRIVATE -UNDEBUG)
endif()
get_target_property(scene_type lux::engine::editor::editor_scene TYPE)
if(NOT scene_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "editor_scene must import its shared implementation")
endif()
get_target_property(scene_includes lux::engine::editor::editor_scene INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(scene_links lux::engine::editor::editor_scene INTERFACE_LINK_LIBRARIES)
file(WRITE "${CMAKE_BINARY_DIR}/scene-import.txt" "includes=${scene_includes}\nlinks=${scene_links}\n")
enable_testing()
add_test(NAME installed_editor_scene COMMAND lux_editor_scene_consumer)
