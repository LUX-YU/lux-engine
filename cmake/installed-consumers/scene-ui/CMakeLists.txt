cmake_minimum_required(VERSION 3.24)
project(scene_ui_consumer LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Qualification requires RelWithDebInfo")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CONSUMER_MODE "CPU_UI" CACHE STRING "CPU_UI, CPU_SCENE or GPU_UI")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()

function(consumer target source)
    # Copy maintained regression code, never include Engine source directories.
    configure_file("${source}" "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp" COPYONLY)
    add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp")
    target_link_libraries(${target} PRIVATE ${ARGN})
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()

if(CONSUMER_MODE STREQUAL "CPU_UI")
    find_package(lux-engine-function REQUIRED COMPONENTS ui)
    find_package(imgui REQUIRED COMPONENTS core)
    if(TARGET lux::engine::function::render_client OR TARGET Vulkan::Vulkan)
        message(FATAL_ERROR "CPU UI imported rendering")
    endif()
    consumer(context "${engine_source}/modules/function/ui/test/context.cpp" lux::engine::function::ui imgui::core)
    add_test(NAME installed.cpu_context COMMAND context)
elseif(CONSUMER_MODE STREQUAL "CPU_SCENE")
    find_package(lux-engine-scene REQUIRED COMPONENTS scene_composition)
    find_package(lux-engine-scene-world-loading REQUIRED COMPONENTS scene_world_loading)
    find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS hierarchy transform)
    if(TARGET lux::engine::function::render_client OR TARGET Vulkan::Vulkan)
        message(FATAL_ERROR "CPU Scene/WorldLoading imported rendering")
    endif()
    consumer(driver "${engine_source}/engine/scene/composition/test/driver.cpp" lux::engine::scene::scene_composition)
    consumer(world_loading "${engine_source}/engine/scene/builtin_systems/world_loading/test/world_loading.cpp"
        lux::engine::scene::scene_world_loading lux::engine::simulation::ecs::hierarchy lux::engine::simulation::ecs::transform)
    add_test(NAME installed.scene_driver COMMAND driver)
    add_test(NAME installed.world_loading COMMAND world_loading "${CMAKE_CURRENT_BINARY_DIR}/world")
elseif(CONSUMER_MODE STREQUAL "GPU_UI")
    find_package(lux-engine-function REQUIRED COMPONENTS ui_rendering)
    find_package(imgui REQUIRED COMPONENTS core)
    if(TARGET lux::engine::editor::editor_ui OR TARGET lux::engine::scene::scene_composition)
        message(FATAL_ERROR "UiRenderFeature imported Editor or Scene")
    endif()
    consumer(feature "${CMAKE_CURRENT_SOURCE_DIR}/feature.cpp" lux::engine::function::ui_rendering imgui::core)
    add_test(NAME installed.ui_feature COMMAND feature)
else()
    message(FATAL_ERROR "Unknown consumer mode")
endif()
