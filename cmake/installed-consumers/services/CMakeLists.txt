cmake_minimum_required(VERSION 3.24)
project(services_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CONSUMER_MODE "CORE" CACHE STRING "CORE, TASKS, AUTHORS, PERSISTENCE, COMPOSITION, or MODEL_PLACEMENT")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
if(CONSUMER_MODE STREQUAL "CORE")
    find_package(lux-engine-core REQUIRED COMPONENTS services)
    find_package(lux-engine-platform REQUIRED COMPONENTS dynamic_library)
    foreach(file services.cpp ServiceProbe.hpp plugin.cpp plugin_module.cpp)
        configure_file("${engine_source}/modules/core/services/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
    endforeach()
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/services.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::core::services)
    add_library(services_plugin MODULE "${CMAKE_BINARY_DIR}/plugin_module.cpp")
    target_link_libraries(services_plugin PRIVATE lux::engine::core::services)
    add_executable(services_plugin_consumer "${CMAKE_BINARY_DIR}/plugin.cpp")
    target_link_libraries(services_plugin_consumer PRIVATE
        lux::engine::core::services lux::engine::platform::dynamic_library)
    set(consumers services_consumer services_plugin services_plugin_consumer)
    add_test(NAME installed.services.core COMMAND services_consumer)
    add_test(NAME installed.services.plugin COMMAND services_plugin_consumer $<TARGET_FILE:services_plugin>)
    foreach(forbidden process_execution ui editor_contracts)
        if(TARGET lux::engine::process::${forbidden} OR TARGET lux::engine::editor::${forbidden}
            OR TARGET lux::engine::function::${forbidden})
            message(FATAL_ERROR "Pure services imported ${forbidden}")
        endif()
    endforeach()
elseif(CONSUMER_MODE STREQUAL "TASKS")
    find_package(lux-engine-core REQUIRED COMPONENTS services)
    find_package(lux-engine-process-execution REQUIRED COMPONENTS process_execution)
    configure_file("${engine_source}/engine/context/test/service_tasks.cpp" "${CMAKE_BINARY_DIR}/tasks.cpp" COPYONLY)
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/tasks.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::core::services lux::engine::process::process_execution)
    set(consumers services_consumer)
    add_test(NAME installed.services.tasks COMMAND services_consumer)
elseif(CONSUMER_MODE STREQUAL "AUTHORS")
    find_package(lux-engine-editor-scene-model REQUIRED COMPONENTS scene_model)
    find_package(lux-engine-editor-material-model REQUIRED COMPONENTS material_model)
    find_package(lux-engine-editor-flowforge-model REQUIRED COMPONENTS flowforge_model)
    configure_file("${engine_source}/editor/tests/architecture/three_sessions.cpp" "${CMAKE_BINARY_DIR}/authors.cpp" COPYONLY)
    configure_file("${engine_source}/cmake/test-support/ObjectQueue.hpp" "${CMAKE_BINARY_DIR}/ObjectQueue.hpp" COPYONLY)
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/authors.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::editor::scene_model
        lux::engine::editor::material_model lux::engine::editor::flowforge_model)
    set(consumers services_consumer)
    add_test(NAME installed.services.authors COMMAND services_consumer)
elseif(CONSUMER_MODE STREQUAL "MODEL_PLACEMENT")
    find_package(lux-engine-toolchain-material REQUIRED COMPONENTS toolchain_material_cooker)
    find_package(lux-engine-editor-scene-execution REQUIRED COMPONENTS scene_project)
    find_package(lux-engine-editor-scene-projection REQUIRED COMPONENTS scene_projection)
    find_package(lux-engine-editor-storage REQUIRED COMPONENTS editor_storage)
    find_package(lux-engine-core REQUIRED COMPONENTS services)
    configure_file("${engine_source}/editor/activities/scene/test/model_placement.cpp"
        "${CMAKE_BINARY_DIR}/model_placement.cpp" COPYONLY)
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/model_placement.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::editor::scene_project lux::engine::editor::scene_projection
        lux::engine::editor::editor_storage lux::engine::core::services lux::engine::toolchain::toolchain_material_cooker)
    set(consumers services_consumer)
    add_test(NAME installed.model_placement COMMAND services_consumer "${CMAKE_BINARY_DIR}/placements")
    foreach(forbidden view_api view_host desktop_shell scene_ui material_ui flow_ui editor_extensions editor_bootstrap)
        if(TARGET lux::engine::editor::${forbidden})
            message(FATAL_ERROR "Headless placement imported ${forbidden}")
        endif()
    endforeach()
elseif(CONSUMER_MODE STREQUAL "PERSISTENCE")
    find_package(lux-engine-editor-material-persistence REQUIRED COMPONENTS material_persistence)
    find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence_execution)
    find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
    find_package(lux-engine-editor-session-factories REQUIRED COMPONENTS session_execution)
    configure_file("${engine_source}/editor/tests/persistence/services.cpp" "${CMAKE_BINARY_DIR}/persistence.cpp" COPYONLY)
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/persistence.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::editor::material_persistence
        lux::engine::editor::editor_persistence_execution lux::engine::editor::editor_file_publication
        lux::engine::editor::session_execution)
    set(consumers services_consumer)
    add_test(NAME installed.services.persistence COMMAND services_consumer "${CMAKE_BINARY_DIR}/files")
    foreach(forbidden editor_composition desktop_shell scene_ui material_ui flowforge_ui editor_bootstrap)
        if(TARGET lux::engine::editor::${forbidden})
            message(FATAL_ERROR "Persistence activities imported ${forbidden}")
        endif()
    endforeach()
elseif(CONSUMER_MODE STREQUAL "COMPOSITION")
    find_package(lux-engine-editor-composition REQUIRED COMPONENTS editor_composition)
    find_package(lux-engine-process-execution REQUIRED COMPONENTS process_execution)
    find_package(lux-engine-platform REQUIRED COMPONENTS dynamic_library)
    foreach(file composition.cpp CompositionProbe.hpp composition_plugin.cpp composition_dll.cpp)
        configure_file("${engine_source}/editor/workbench/desktop/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
    endforeach()
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/composition.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::editor::editor_composition)
    add_library(services_plugin MODULE "${CMAKE_BINARY_DIR}/composition_plugin.cpp")
    target_link_libraries(services_plugin PRIVATE lux::engine::editor::editor_composition
        lux::engine::process::process_execution)
    add_executable(services_plugin_consumer "${CMAKE_BINARY_DIR}/composition_dll.cpp")
    target_link_libraries(services_plugin_consumer PRIVATE lux::engine::editor::editor_composition
        lux::engine::process::process_execution lux::engine::platform::dynamic_library)
    set(consumers services_consumer services_plugin services_plugin_consumer)
    add_test(NAME installed.composition COMMAND services_consumer)
    add_test(NAME installed.composition.dll COMMAND services_plugin_consumer $<TARGET_FILE:services_plugin>)
    foreach(forbidden view_api view_host desktop_shell scene_ui material_ui flow_ui editor_extensions editor_bootstrap)
        if(TARGET lux::engine::editor::${forbidden})
            message(FATAL_ERROR "Neutral composition imported ${forbidden}")
        endif()
    endforeach()
else()
    message(FATAL_ERROR "Unknown CONSUMER_MODE")
endif()
foreach(consumer IN LISTS consumers)
    if(MSVC)
        target_compile_options(${consumer} PRIVATE /UNDEBUG /permissive- /utf-8)
    else()
        target_compile_options(${consumer} PRIVATE -UNDEBUG)
    endif()
endforeach()
