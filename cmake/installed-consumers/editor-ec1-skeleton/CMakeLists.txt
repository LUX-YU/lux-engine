cmake_minimum_required(VERSION 3.24)
project(ec1_skeleton_sdk LANGUAGES CXX)
if(NOT LUX_TEST_SUPPORT_DIR)
    get_filename_component(LUX_TEST_SUPPORT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../test-support" ABSOLUTE)
endif()
include_directories("${LUX_TEST_SUPPORT_DIR}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT SDK_PREFIX)
    message(FATAL_ERROR "SDK_PREFIX must be a standalone installed SDK")
endif()
set(EC1_MODE "HEADLESS" CACHE STRING "HEADLESS, WINDOW or APP")
include("${SDK_PREFIX}/share/lux-engine/plugins/LuxPlugins.cmake")
find_package(lux-engine-editor-extensions REQUIRED COMPONENTS editor_extensions)
find_package(lux-engine-editor-storage REQUIRED COMPONENTS editor_storage)
find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence_execution)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
find_package(lux-engine-editor-workspace-store REQUIRED COMPONENTS workspace_store)
find_package(lux-engine-editor-desktop REQUIRED COMPONENTS view_host)
add_library(skeleton_runtime SHARED Runtime.cpp)
target_link_libraries(skeleton_runtime PRIVATE lux::engine::editor::editor_extensions)
add_library(skeleton_extension SHARED Model.cpp Extension.cpp)
target_link_libraries(skeleton_extension PRIVATE lux::engine::editor::editor_extensions)
lux_add_plugin_exports(TARGET skeleton_runtime EDITOR_TARGET skeleton_extension DESCRIPTION "${CMAKE_CURRENT_SOURCE_DIR}/plugin.json"
    INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/Runtime.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/Model.cpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/Model.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/Extension.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/Probe.hpp"
        "${CMAKE_CURRENT_SOURCE_DIR}/Settings.hpp")
add_executable(skeleton_consumer main.cpp)
target_link_libraries(skeleton_consumer PRIVATE lux::engine::editor::editor_extensions
    lux::engine::editor::editor_storage lux::engine::editor::editor_persistence_execution
    lux::engine::editor::editor_file_publication lux::engine::editor::workspace_store
    lux::engine::editor::view_host)
if(EC1_MODE STREQUAL "APP")
    add_library(witness_runtime SHARED Runtime.cpp)
    target_link_libraries(witness_runtime PRIVATE lux::engine::editor::editor_extensions)
    add_library(witness_extension SHARED Witness.cpp)
    target_link_libraries(witness_extension PRIVATE lux::engine::editor::editor_extensions)
    lux_add_plugin_exports(TARGET witness_runtime EDITOR_TARGET witness_extension
        DESCRIPTION "${CMAKE_CURRENT_SOURCE_DIR}/witness.json"
        INPUTS "${CMAKE_CURRENT_SOURCE_DIR}/Witness.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/Probe.hpp")
    find_package(lux-engine-editor-application REQUIRED COMPONENTS editor_bootstrap)
    target_compile_definitions(skeleton_consumer PRIVATE EC1_APP=1)
    target_link_libraries(skeleton_consumer PRIVATE lux::engine::editor::editor_bootstrap)
elseif(NOT EC1_MODE MATCHES "^(HEADLESS|WINDOW)$")
    message(FATAL_ERROR "Unknown EC1_MODE")
endif()
if(NOT EC1_MODE STREQUAL "APP")
    foreach(forbidden editor_bootstrap editor_builtin_contributions material_ui flow_ui scene_ui renderer)
        if(TARGET lux::engine::editor::${forbidden})
            message(FATAL_ERROR "CPU consumer imported ${forbidden}")
        endif()
    endforeach()
endif()
set(example_targets skeleton_runtime skeleton_extension skeleton_consumer)
if(EC1_MODE STREQUAL "APP")
    list(APPEND example_targets witness_runtime witness_extension)
endif()
foreach(target IN LISTS example_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    if(WIN32)
        add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
            $<TARGET_RUNTIME_DLLS:${target}> "$<TARGET_FILE_DIR:${target}>" COMMAND_EXPAND_LISTS)
    endif()
endforeach()
enable_testing()
add_test(NAME installed.ec1.skeleton.${EC1_MODE} COMMAND skeleton_consumer
    "${CMAKE_BINARY_DIR}" "${SDK_PREFIX}" "${EC1_MODE}")
