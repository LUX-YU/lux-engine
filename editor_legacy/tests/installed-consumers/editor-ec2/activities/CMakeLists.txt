cmake_minimum_required(VERSION 3.24)
project(ec2_editor_activities LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(EC2_MODE "ACTIVITIES" CACHE STRING "ACTIVITIES or PANELS")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
enable_testing()
function(qualification name source)
    configure_file("${engine_source}/${source}" "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp" COPYONLY)
    add_executable(${name} "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    target_link_libraries(${name} PRIVATE ${ARGN})
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
endfunction()
if(EC2_MODE STREQUAL "ACTIVITIES")
    find_package(lux-engine-editor-scene-execution REQUIRED COMPONENTS scene_execution)
    find_package(lux-engine-project-plugin-rendering REQUIRED COMPONENTS project_plugin_rendering)
    find_package(lux-engine-editor-storage REQUIRED COMPONENTS editor_storage)
    find_package(lux-engine-editor-assets REQUIRED COMPONENTS editor_assets)
    find_package(lux-engine-editor-material-persistence REQUIRED COMPONENTS material_persistence)
    find_package(lux-engine-editor-material-preview REQUIRED COMPONENTS material_preview)
    find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
    qualification(configuration editor/activities/scene/test/configuration.cpp
        lux::engine::editor::scene_execution lux::engine::project::project_plugin_rendering)
    qualification(publication editor/tests/integration/project/artifact_publication.cpp
        lux::engine::editor::editor_storage lux::engine::editor::editor_file_publication
        lux::engine::editor::material_preview lux::engine::editor::material_persistence)
    qualification(recipe editor/tests/integration/project/model_recipe.cpp lux::engine::editor::editor_assets)
    add_test(NAME installed.ec2.configuration COMMAND configuration "${SDK_PREFIX}")
    add_test(NAME installed.ec2.publication COMMAND publication "${CMAKE_CURRENT_BINARY_DIR}/files")
    add_test(NAME installed.ec2.recipe COMMAND recipe)
    if(TARGET lux::engine::function::ui OR TARGET lux::engine::editor::scene_ui)
        message(FATAL_ERROR "CPU configuration/IO consumer imported UI")
    endif()
else()
    if(NOT EC2_MODE STREQUAL "PANELS")
        message(FATAL_ERROR "Unknown EC2_MODE")
    endif()
    find_package(lux-engine-editor-project-tools-ui REQUIRED COMPONENTS project_tools_ui)
    qualification(panels editor/tests/integration/project/panels.cpp lux::engine::editor::project_tools_ui)
    file(READ "${CMAKE_CURRENT_BINARY_DIR}/panels.cpp" code)
    string(REPLACE "../../../../cmake/installed-consumers/common/UiTestContent.hpp" "UiTestContent.hpp" code "${code}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/panels.cpp" "${code}")
    configure_file("${engine_source}/cmake/installed-consumers/common/UiTestContent.hpp"
        "${CMAKE_CURRENT_BINARY_DIR}/UiTestContent.hpp" COPYONLY)
    add_test(NAME installed.ec2.panels COMMAND panels)
endif()
