cmake_minimum_required(VERSION 3.24)
project(lux_object_core_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(lux-engine-core REQUIRED COMPONENTS object)
if(TARGET lux::engine::core::meta OR TARGET lux::engine::core::object_reflection)
    message(FATAL_ERROR "Object core imported the optional reflection closure")
endif()
add_executable(object_core_consumer main.cpp)
target_link_libraries(object_core_consumer PRIVATE lux::engine::core::object)
if(MSVC)
    target_compile_options(object_core_consumer PRIVATE /UNDEBUG)
else()
    target_compile_options(object_core_consumer PRIVATE -UNDEBUG)
endif()
