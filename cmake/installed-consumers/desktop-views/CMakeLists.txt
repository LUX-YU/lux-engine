cmake_minimum_required(VERSION 3.24)
project(desktop_views_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(lux-engine-editor-desktop REQUIRED COMPONENTS view_host desktop_shell)
find_package(lux-engine-editor-scene-ui REQUIRED COMPONENTS scene_ui)
find_package(lux-engine-editor-material-ui REQUIRED COMPONENTS material_ui)
find_package(lux-engine-editor-flow-ui REQUIRED COMPONENTS flow_ui)
find_package(lux-engine-editor-tasks-ui REQUIRED COMPONENTS tasks_ui)
find_package(lux-engine-editor-project-io REQUIRED COMPONENTS project_io)
find_package(lux-engine-editor-material-persistence REQUIRED COMPONENTS material_persistence)
find_package(lux-engine-editor-scene-execution REQUIRED COMPONENTS scene_execution)
find_package(lux-engine-function REQUIRED COMPONENTS render_features ui_rendering)
find_package(lux-engine-scene-transform REQUIRED COMPONENTS scene_transform)
find_package(imgui REQUIRED COMPONENTS core)
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-cxx REQUIRED COMPONENTS reflection_generator)
find_package(lux-engine-core REQUIRED COMPONENTS meta serialization)
find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS schema)
foreach(forbidden editor_ui editor_scene editor_material editor_flowforge editor_context editor_storage editor_app)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "New desktop SDK imported legacy target ${forbidden}")
    endif()
endforeach()
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LUX_GENERATE_HEADER_DIR "${CMAKE_BINARY_DIR}/generated")
include_component_cmake_scripts(lux::engine::core::meta)
include_component_cmake_scripts(lux::engine::simulation::ecs::schema)
include_component_cmake_scripts(lux::engine::editor::scene_ui)
set(component_input "${CMAKE_CURRENT_BINARY_DIR}/include/consumer/Component.hpp")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../editor-d2/include/consumer/Component.hpp" "${component_input}" COPYONLY)
add_executable(inspector_fields Inspector.cpp)
set_property(TARGET inspector_fields PROPERTY LUX_ARCH_LAYER EDITOR)
target_include_directories(inspector_fields PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/include" "${LUX_GENERATE_HEADER_DIR}")
target_link_libraries(inspector_fields PRIVATE lux::engine::editor::scene_ui lux::engine::editor::view_host)
engine_target_add_ecs_component_codegen(NAME desktop_consumer_schema TARGET inspector_fields
    SOURCE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/Inspector.cpp" HEADER "${component_input}"
    LOGICAL_PATH consumer/Component.hpp SYMBOL DesktopConsumer)
engine_target_add_imgui_inspector_codegen(NAME desktop_consumer TARGET inspector_fields SESSION_FIELDS
    SOURCE_FILE "${CMAKE_CURRENT_SOURCE_DIR}/Inspector.cpp" HEADER "${component_input}"
    LOGICAL_PATH consumer/Component.hpp COMPONENTS consumer::Component)
# Only maintained test code is copied. All production headers and libraries come from the installed SDK.
file(READ "${engine_source}/editor/tools/scene/ui/test/views.cpp" source)
string(REPLACE "../../../../../cmake/installed-consumers/common/ControlsTestAccess.hpp"
    "ControlsTestAccess.hpp" source "${source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/views.cpp" "${source}")
foreach(helper NativeDesktop AuxiliaryViews)
    configure_file("${engine_source}/editor/tools/scene/ui/test/${helper}.hpp"
        "${CMAKE_CURRENT_BINARY_DIR}/${helper}.hpp" COPYONLY)
endforeach()
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../common/ControlsTestAccess.hpp"
    "${CMAKE_CURRENT_BINARY_DIR}/ControlsTestAccess.hpp" COPYONLY)
configure_file("${engine_source}/editor/desktop/test/host.cpp" "${CMAKE_CURRENT_BINARY_DIR}/host.cpp" COPYONLY)
add_executable(desktop_views "${CMAKE_CURRENT_BINARY_DIR}/views.cpp")
target_link_libraries(desktop_views PRIVATE lux::engine::editor::scene_ui lux::engine::editor::desktop_shell
    lux::engine::editor::material_ui lux::engine::editor::flow_ui lux::engine::editor::scene_execution
    lux::engine::editor::tasks_ui lux::engine::editor::project_io lux::engine::editor::material_persistence
    lux::engine::scene::scene_transform lux::engine::function::render_features
    lux::engine::function::ui_rendering imgui::core)
add_executable(view_host "${CMAKE_CURRENT_BINARY_DIR}/host.cpp")
target_link_libraries(view_host PRIVATE lux::engine::editor::view_host)
foreach(target desktop_views view_host inspector_fields)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
find_program(P10_LINKER NAMES lld-link REQUIRED)
get_target_property(scene_ui_includes lux::engine::editor::scene_ui INTERFACE_INCLUDE_DIRECTORIES)
list(GET scene_ui_includes 0 sdk_include)
get_filename_component(sdk_prefix "${sdk_include}" DIRECTORY)
if(NOT EXISTS "${sdk_prefix}/share/lux-engine/plugins/catalog.json")
    message(FATAL_ERROR "The installed SDK must include its plugin catalog: ${sdk_prefix}")
endif()
enable_testing()
add_test(NAME installed.desktop.host COMMAND view_host)
add_test(NAME installed.desktop.inspector_codegen COMMAND inspector_fields)
add_test(NAME installed.desktop.real_views COMMAND desktop_views "${P10_LINKER}" "${sdk_prefix}")
set_tests_properties(installed.desktop.real_views PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
if(WIN32)
    target_link_libraries(desktop_views PRIVATE user32)
    add_test(NAME installed.desktop.native_input COMMAND desktop_views "${P10_LINKER}" "${sdk_prefix}" native)
    set_tests_properties(installed.desktop.native_input PROPERTIES TIMEOUT 120 RUN_SERIAL TRUE)
endif()
