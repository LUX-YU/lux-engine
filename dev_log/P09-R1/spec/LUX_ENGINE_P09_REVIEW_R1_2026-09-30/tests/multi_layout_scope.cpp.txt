// Insert into editor/tests/workspace/workspace.cpp's anonymous namespace.
// Uses that file's actual Fixture, writeRaw, raw, take, w, v, fs helpers.
// Register separate scenario names in its main/CMake and the installed workspace consumer.
// Do not run against a user's project. Fixture root must be a test-owned temporary directory.

std::string legacyMaterial(std::string_view pane, std::string_view asset)
{
    std::string text = "version = 1\ndock = '''[Window][Material###";
    text += pane;
    text += "]\nPos=0,0\nSize=800,600\n'''\n[[panes]]\nid = '";
    text += pane;
    text += "'\ntype = 'lux.editor.material.v1'\nvisible = true\npayload = 'v1:";
    text += asset;
    text += "'\n";
    return text;
}

// same_slot=true: current code is inferred to reject before settings is read.
// same_slot=false: current code is inferred to merge the nonselected locator.
// Tests must first collect actual results on the unmodified P09 implementation.
bool multiLayoutScope(Fixture& f, bool same_slot, bool select_beta)
{
    constexpr std::string_view a = "12345678-1234-1234-1234-123456789abc";
    constexpr std::string_view b = "12345678-1234-1234-1234-123456789abd";
    const std::string beta_pane = same_slot ? "material-1" : "material-2";
    const auto alpha = legacyMaterial("material-1", a);
    const auto beta = legacyMaterial(beta_pane, b);
    const auto old = f.root / ".lux/editor/layouts";
    writeRaw(old / "Alpha.toml", alpha);
    writeRaw(old / "Beta.toml", beta);
    writeRaw(f.root / ".lux/editor/settings.toml",
             select_beta ? "version=1\nselected='Beta'\n" : "version=1\nselected='Alpha'\n");
    const auto before_size = f.coordinator.size();
    auto result = f.store.prepareLegacyMigration();
    if (!result)
    {
        std::printf("multi-layout accepted=0 error=%u detail=%s\n",
                    unsigned(result.error().code), result.error().detail.c_str());
        std::fflush(stdout);
        assert(raw(old / "Alpha.toml") == alpha && raw(old / "Beta.toml") == beta);
        assert(f.coordinator.size() == before_size);
        return false; // Desired contract: these two individually legal records are not a global collision.
    }
    auto migration = std::move(*result);
    const auto expected = std::string("asset:") + std::string(select_beta ? b : a);
    const auto expected_key = std::string("legacy:") + (select_beta ? beta_pane : "material-1");
    const bool scoped = migration.recovery.entries.size() == 1 &&
        migration.recovery.entries.front().locator == expected &&
        migration.recovery.entries.front().restore_key.name() == expected_key;
    std::printf("multi-layout accepted=1 layouts=%zu recovery_entries=%zu selected_scope=%d\n",
                migration.layouts.size(), migration.recovery.entries.size(), scoped);
    std::fflush(stdout);
    if (!scoped || migration.layouts.size() != 2)
        return false;
    assert(migration.layouts[0].id != migration.layouts[1].id);
    assert(migration.layouts[0].slots.size() == 1 && migration.layouts[1].slots.size() == 1);
    if (same_slot)
        assert(migration.layouts[0].slots[0].restore_key == migration.layouts[1].slots[0].restore_key);
    assert(f.coordinator.size() == before_size); // prepare is not publication.
    assert(raw(old / "Alpha.toml") == alpha && raw(old / "Beta.toml") == beta);
    // Exercise real publication and restart; repeat with interruption and an edited first target separately.
    unsigned count = 0;
    while (auto ticket = take(f.store.continueMigration(migration)))
    {
        assert(++count <= migration.layouts.size() + 3); // 2 layouts, recovery, preferences, marker.
        f.success(*ticket);
    }
    w::WorkspaceStore restarted(f.root, f.coordinator, f.backend);
    const auto recovery = take(restarted.readRecovery());
    assert(recovery.value.entries.size() == 1 && recovery.value.entries.front().locator == expected);
    assert(take(restarted.listLayouts()).layouts.size() == 2);
    assert(!take(restarted.continueMigration(migration)));
    assert(raw(old / "Alpha.toml") == alpha && raw(old / "Beta.toml") == beta);
    return true;
}
