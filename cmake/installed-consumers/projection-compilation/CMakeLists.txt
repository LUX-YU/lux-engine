cmake_minimum_required(VERSION 3.22)
project(projection_compilation_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(lux-engine-editor-material-preview REQUIRED COMPONENTS material_preview)
find_package(lux-engine-editor-flowforge-compilation REQUIRED COMPONENTS flowforge_compilation)
find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence_execution)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
find_package(lux-engine-scene-transform REQUIRED COMPONENTS scene_transform)
if(TARGET lux::engine::editor::editor_ui OR TARGET lux::engine::editor::editor_scene
    OR TARGET lux::engine::editor::editor_material OR TARGET lux::engine::editor::editor_flowforge
    OR TARGET lux::engine::editor::editor_context OR TARGET lux::engine::editor::editor_storage)
    message(FATAL_ERROR "Independent projection/compiler SDK imported a legacy editor")
endif()
add_executable(projection_consumer projection.cpp)
target_link_libraries(projection_consumer PRIVATE lux::engine::editor::scene_projection lux::engine::scene::scene_transform)
add_executable(compilation_consumer compilation.cpp)
target_link_libraries(compilation_consumer PRIVATE lux::engine::editor::material_preview
    lux::engine::editor::flowforge_compilation lux::engine::editor::editor_persistence_execution lux::engine::editor::editor_file_publication)
add_executable(ownership_consumer ownership.cpp)
target_link_libraries(ownership_consumer PRIVATE lux::engine::editor::material_preview
    lux::engine::editor::flowforge_compilation)
foreach(consumer projection_consumer compilation_consumer ownership_consumer)
    if(MSVC)
        target_compile_options(${consumer} PRIVATE /UNDEBUG /utf-8)
    else()
        target_compile_options(${consumer} PRIVATE -UNDEBUG)
    endif()
endforeach()
if(WIN32)
    find_program(P07_LINKER NAMES lld-link REQUIRED)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_program(P07_LINKER NAMES ld.lld ld REQUIRED)
else()
    message(FATAL_ERROR "This Flow AOT consumer requires a qualified COFF or ELF host")
endif()
enable_testing()
add_test(NAME installed.projection.reset_capacity COMMAND projection_consumer)
add_test(NAME installed.compilation.actual_publish COMMAND compilation_consumer "${P07_LINKER}" "${CMAKE_CURRENT_BINARY_DIR}/publication")
foreach(kind material flow)
    add_test(NAME installed.compilation.ownership_${kind} COMMAND ownership_consumer ${kind} "${P07_LINKER}")
    foreach(action copy_construct copy_assign move_construct move_assign)
        set(probe reject_${kind}_${action})
        add_library(${probe} OBJECT EXCLUDE_FROM_ALL ownership_negative.cpp)
        target_link_libraries(${probe} PRIVATE lux::engine::editor::material_preview lux::engine::editor::flowforge_compilation)
        target_compile_definitions(${probe} PRIVATE OWNER_${kind} ACTION_${action})
        add_test(NAME installed.compilation.${probe} COMMAND "${CMAKE_COMMAND}"
            "-DBUILD_DIR=${CMAKE_CURRENT_BINARY_DIR}" "-DPROBE=${probe}" -P "${CMAKE_CURRENT_SOURCE_DIR}/reject_copy.cmake")
        set_tests_properties(installed.compilation.${probe} PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
    endforeach()
endforeach()
