cmake_minimum_required(VERSION 3.27)
project(editor_ec3_workspace LANGUAGES CXX)
find_package(lux-engine-editor-desktop REQUIRED COMPONENTS desktop_shell)
find_package(lux-engine-editor-workspace-store REQUIRED COMPONENTS workspace_store)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
if(TARGET lux::engine::editor::editor_bootstrap)
    message(FATAL_ERROR "Workspace capability must not import Application")
endif()
add_executable(editor_ec3_workspace main.cpp)
target_compile_features(editor_ec3_workspace PRIVATE cxx_std_20)
set_target_properties(editor_ec3_workspace PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(editor_ec3_workspace PRIVATE lux::engine::editor::desktop_shell
    lux::engine::editor::workspace_store lux::engine::editor::editor_file_publication)
if(MSVC)
    target_compile_options(editor_ec3_workspace PRIVATE /UNDEBUG /utf-8 /permissive-)
else()
    target_compile_options(editor_ec3_workspace PRIVATE -UNDEBUG)
endif()
