cmake_minimum_required(VERSION 3.22)
project(lux_script_runtime_input_consumer LANGUAGES CXX)
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-engine-simulation REQUIRED COMPONENTS simulation_script)
get_target_property(runtime_links lux::engine::simulation::simulation_script INTERFACE_LINK_LIBRARIES)
if(runtime_links MATCHES "world::identity|scene_script_description|scene_script_runtime")
    message(FATAL_ERROR "Direct Script runtime exports a persistent/loading dependency: ${runtime_links}")
endif()
add_executable(lux_script_runtime_input_consumer main.cpp)
target_compile_features(lux_script_runtime_input_consumer PRIVATE cxx_std_20)
target_compile_options(lux_script_runtime_input_consumer PRIVATE /UNDEBUG)
target_link_libraries(lux_script_runtime_input_consumer PRIVATE lux::engine::simulation::simulation_script)
