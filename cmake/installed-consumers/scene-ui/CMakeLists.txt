cmake_minimum_required(VERSION 3.24)
project(scene_ui_consumer LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Qualification requires RelWithDebInfo")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CONSUMER_MODE "CPU_UI" CACHE STRING "CPU_UI, CPU_SCENE, GPU_UI or EDITOR_SCENE_PANE")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()

function(consumer target source)
    # Copy maintained regression code, never include Engine source directories.
    file(READ "${source}" consumer_source)
    string(REGEX REPLACE "#include \"[^\"]*/common/(UiDrawPane|UiTestContent|RenderRegistration)\\.hpp\""
        "#include \"\\1.hpp\"" consumer_source "${consumer_source}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp" "${consumer_source}")
    foreach(helper UiDrawPane UiTestContent RenderRegistration)
        configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../common/${helper}.hpp"
            "${CMAKE_CURRENT_BINARY_DIR}/${helper}.hpp" COPYONLY)
    endforeach()
    add_executable(${target} "${CMAKE_CURRENT_BINARY_DIR}/${target}.cpp")
    target_link_libraries(${target} PRIVATE ${ARGN})
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()

if(CONSUMER_MODE STREQUAL "CPU_UI")
    find_package(lux-engine-function REQUIRED COMPONENTS ui)
    find_package(imgui REQUIRED COMPONENTS core)
    if(TARGET lux::engine::function::render_vulkan OR TARGET lux::engine::function::render_runtime
        OR TARGET lux::engine::function::ui_rendering OR TARGET Vulkan::Vulkan)
        message(FATAL_ERROR "CPU UI imported GPU execution")
    endif()
    consumer(context "${engine_source}/modules/function/ui/test/context.cpp" lux::engine::function::ui imgui::core)
    add_test(NAME installed.cpu_context COMMAND context)
elseif(CONSUMER_MODE STREQUAL "CPU_SCENE")
    find_package(lux-engine-scene REQUIRED COMPONENTS scene_composition)
    find_package(lux-engine-scene-world-loading REQUIRED COMPONENTS scene_world_loading)
    find_package(lux-engine-scene-transform REQUIRED COMPONENTS scene_transform)
    find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS hierarchy transform)
    if(TARGET lux::engine::function::render_client OR TARGET Vulkan::Vulkan)
        message(FATAL_ERROR "CPU Scene/WorldLoading imported rendering")
    endif()
    consumer(driver "${engine_source}/engine/scene/composition/test/runtime.cpp" lux::engine::scene::scene_composition lux::engine::scene::scene_transform)
    consumer(world_loading "${engine_source}/engine/scene/builtin_systems/world_loading/test/world_loading.cpp"
        lux::engine::scene::scene_world_loading lux::engine::simulation::ecs::hierarchy lux::engine::simulation::ecs::transform)
    add_test(NAME installed.scene_runtime COMMAND driver)
    add_test(NAME installed.world_loading COMMAND world_loading "${CMAKE_CURRENT_BINARY_DIR}/world")
elseif(CONSUMER_MODE STREQUAL "GPU_UI")
    find_package(lux-engine-function REQUIRED COMPONENTS ui_rendering)
    find_package(imgui REQUIRED COMPONENTS core)
    if(TARGET lux::engine::function::render_runtime)
        message(FATAL_ERROR "UI rendering imported a Runtime owner")
    endif()
    if(TARGET lux::engine::editor::editor_ui OR TARGET lux::engine::scene::scene_composition)
        message(FATAL_ERROR "RenderFeature imported Editor or Scene")
    endif()
    # This executable owns a device/thread; the UI library itself does not.
    find_package(lux-engine-function REQUIRED COMPONENTS render_runtime)
    consumer(feature "${CMAKE_CURRENT_SOURCE_DIR}/feature.cpp"
        lux::engine::function::ui_rendering lux::engine::function::render_runtime imgui::core)
    add_test(NAME installed.ui_feature COMMAND feature)
elseif(CONSUMER_MODE STREQUAL "EDITOR_SCENE_PANE")
    find_package(lux-engine-editor-ui REQUIRED COMPONENTS editor_ui)
    find_package(lux-engine-function REQUIRED COMPONENTS render_features)
    find_package(lux-engine-platform REQUIRED COMPONENTS window)
    if(TARGET lux::engine::editor::editor_scene OR TARGET lux::engine::editor::editor_material)
        message(FATAL_ERROR "Public ViewportElement imported a concrete editor")
    endif()
    consumer(scene_panes "${engine_source}/editor/ui/test/unified_ui.cpp"
        lux::engine::editor::editor_ui lux::engine::platform::window lux::engine::function::render_features)
    add_test(NAME installed.scene_panes COMMAND scene_panes --scene-panes)
else()
    message(FATAL_ERROR "Unknown consumer mode")
endif()
