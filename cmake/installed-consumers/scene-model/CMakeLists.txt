cmake_minimum_required(VERSION 3.25)
project(lux_scene_model_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(lux-engine-editor-scene-model REQUIRED COMPONENTS scene_model)
add_executable(scene_model_consumer main.cpp)
target_link_libraries(scene_model_consumer PRIVATE lux::engine::editor::scene_model)
if(MSVC)
    target_compile_options(scene_model_consumer PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor)
else()
    target_compile_options(scene_model_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.scene_model COMMAND scene_model_consumer)
