cmake_minimum_required(VERSION 3.22)
project(lux_editor_ui_consumer LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(lux-engine-editor-ui REQUIRED COMPONENTS editor_ui)
find_package(lux-engine-platform REQUIRED COMPONENTS window)
add_executable(editor_ui_consumer main.cpp)
target_compile_features(editor_ui_consumer PRIVATE cxx_std_20)
target_link_libraries(editor_ui_consumer PRIVATE lux::engine::editor::editor_ui lux::engine::platform::window)
target_compile_options(editor_ui_consumer PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG>)
get_target_property(ui_links lux::engine::editor::editor_ui INTERFACE_LINK_LIBRARIES)
if(ui_links MATCHES "scene|rendering|material|flowforge|context")
    message(FATAL_ERROR "Generic Editor UI exports a business dependency: ${ui_links}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/editor-ui-import.txt" "links=${ui_links}\n")
enable_testing()
add_test(NAME installed_generic_editor_ui COMMAND editor_ui_consumer)
set(LUX_TEST_FONT "" CACHE FILEPATH "Explicit external font input; never supplied by the SDK")
if(LUX_TEST_FONT)
    add_test(NAME installed_editor_ui_external_font COMMAND editor_ui_consumer "${LUX_TEST_FONT}")
endif()
