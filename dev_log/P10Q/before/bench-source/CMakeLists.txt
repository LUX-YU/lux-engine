cmake_minimum_required(VERSION 3.25)
project(p10q_sdk_measurements LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-project-ui REQUIRED COMPONENTS project_ui)
find_package(lux-engine-editor-tasks-ui REQUIRED COMPONENTS tasks_ui)
find_package(lux-engine-editor-material-preview REQUIRED COMPONENTS material_preview)
add_executable(measure main.cpp)
target_link_libraries(measure PRIVATE lux::engine::editor::project_ui lux::engine::editor::tasks_ui
    lux::engine::editor::material_preview)
if(QUALITY_AFTER)
 target_compile_definitions(measure PRIVATE QUALITY_AFTER=1)
endif()
if(MSVC)
 target_compile_options(measure PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
endif()
