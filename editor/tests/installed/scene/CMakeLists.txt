cmake_minimum_required(VERSION 3.28)
project(SceneFileConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-framework REQUIRED COMPONENTS lux_editor_scene)
add_executable(scene_consumer ../../scene_file.cpp)
target_link_libraries(scene_consumer PRIVATE lux::engine::editor::lux_editor_scene)
if(MSVC)
    target_compile_options(scene_consumer PRIVATE /UNDEBUG /utf-8 /GR- /we4541 /permissive-)
else()
    target_compile_options(scene_consumer PRIVATE -UNDEBUG -fno-rtti)
endif()
enable_testing()
add_test(NAME sdk.scene_file COMMAND scene_consumer "${CMAKE_CURRENT_BINARY_DIR}/data")
