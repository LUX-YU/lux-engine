cmake_minimum_required(VERSION 3.22)
project(lux_sdk_relocation_probe LANGUAGES NONE)
if(NOT DEFINED EXPECTED_PREFIX)
    message(FATAL_ERROR "EXPECTED_PREFIX is required")
endif()
find_package(lux-cxx REQUIRED CONFIG COMPONENTS container)
get_target_property(actual_include lux::cxx::container INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(actual_assets lux::cxx::container IMPORTED_ASSETS_PREFIX)
file(TO_CMAKE_PATH "${EXPECTED_PREFIX}" expected)
file(WRITE "${CMAKE_BINARY_DIR}/resolved-paths.txt"
    "expected_prefix=${expected}\ninclude=${actual_include}\nassets=${actual_assets}\n")
foreach(path IN LISTS actual_include actual_assets)
    file(TO_CMAKE_PATH "${path}" normalized)
    string(FIND "${normalized}" "${expected}/" prefix_position)
    if(NOT prefix_position EQUAL 0)
        message(FATAL_ERROR
            "Relocated SDK still references another prefix: ${normalized}; expected ${expected}")
    endif()
endforeach()
