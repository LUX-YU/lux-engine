cmake_minimum_required(VERSION 3.24)
project(editor_ec3_declarations LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-commands REQUIRED COMPONENTS editor_commands)
find_package(lux-engine-editor-session-factories REQUIRED COMPONENTS session_factories)
find_package(lux-engine-editor-composition REQUIRED COMPONENTS editor_composition)
foreach(forbidden view_api view_host)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "Declaration consumer imported retired protocol ${forbidden}")
    endif()
endforeach()
add_executable(editor_ec3_declarations main.cpp)
target_link_libraries(editor_ec3_declarations PRIVATE lux::engine::editor::editor_commands lux::engine::editor::session_factories lux::engine::editor::editor_composition)
if(MSVC)
    target_compile_options(editor_ec3_declarations PRIVATE /UNDEBUG /permissive- /utf-8)
else()
    target_compile_options(editor_ec3_declarations PRIVATE -UNDEBUG)
endif()
if(DEFINED EC3_INVALID_DECLARATION)
    target_compile_definitions(editor_ec3_declarations PRIVATE "EC3_INVALID_DECLARATION=${EC3_INVALID_DECLARATION}")
endif()
option(EC3_MEASURE_COMMANDS "Use native-test SDK counters in the real command library" OFF)
if(EC3_MEASURE_COMMANDS)
    find_package(lux-engine-editor-desktop REQUIRED COMPONENTS desktop_shell)
    add_executable(editor_ec3_measure measure.cpp)
    target_link_libraries(editor_ec3_measure PRIVATE lux::engine::editor::editor_commands lux::engine::editor::desktop_shell)
    if(MSVC)
        target_compile_options(editor_ec3_measure PRIVATE /UNDEBUG /permissive- /utf-8)
    else()
        target_compile_options(editor_ec3_measure PRIVATE -UNDEBUG)
    endif()
endif()
