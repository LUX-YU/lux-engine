cmake_minimum_required(VERSION 3.24)
project(editor_p11_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT SDK_PREFIX)
    message(FATAL_ERROR "SDK_PREFIX must select the installed SDK")
endif()
include("${SDK_PREFIX}/share/lux-engine/plugins/LuxPlugins.cmake")
find_package(lux-engine-project-plugins REQUIRED COMPONENTS project_plugins)
find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS transform)
find_package(lux-engine-editor-extensions REQUIRED COMPONENTS editor_extensions)
find_package(lux-engine-editor-desktop REQUIRED COMPONENTS desktop_shell)
find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence_execution)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
foreach(legacy editor_context editor_metadata editor_ui editor_app editor_scene editor_material editor_flowforge)
    if(TARGET lux::engine::editor::${legacy})
        message(FATAL_ERROR "Formal extension imported ${legacy}")
    endif()
endforeach()
configure_file(Identity.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp" @ONLY)
add_library(p11_runtime SHARED Runtime.cpp "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp")
target_link_libraries(p11_runtime PRIVATE lux::engine::simulation::ecs::transform)
add_library(p11_extension SHARED Extension.cpp "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp")
target_link_libraries(p11_extension PRIVATE lux::engine::editor::editor_extensions)
foreach(mode RANGE 6 10)
    add_library(p11_reject_${mode} SHARED Rejected.cpp "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp")
    target_link_libraries(p11_reject_${mode} PRIVATE lux::engine::editor::editor_extensions)
    target_compile_definitions(p11_reject_${mode} PRIVATE PROBE_REJECT=${mode})
endforeach()
add_executable(p11_consumer main.cpp)
target_link_libraries(p11_consumer PRIVATE lux::engine::editor::editor_extensions lux::engine::editor::view_host
    lux::engine::editor::editor_persistence_execution lux::engine::editor::editor_file_publication)
get_property(targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(target IN LISTS targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
enable_testing()
foreach(mode IN ITEMS with-configuration without-configuration)
add_test(NAME installed.p11.extension.${mode} COMMAND p11_consumer
    "$<TARGET_FILE:p11_runtime>" "$<TARGET_FILE:p11_extension>"
    "$<TARGET_FILE:p11_reject_6>" "$<TARGET_FILE:p11_reject_7>"
    "$<TARGET_FILE:p11_reject_8>" "$<TARGET_FILE:p11_reject_9>" "$<TARGET_FILE:p11_reject_10>" "${mode}")
endforeach()
