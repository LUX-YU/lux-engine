cmake_minimum_required(VERSION 3.24)
project(d3_owner_check LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
  message(FATAL_ERROR "RelWithDebInfo required")
endif()
find_package(lux-engine-function REQUIRED COMPONENTS render_vulkan)
find_package(Vulkan REQUIRED)
get_filename_component(D3_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
add_executable(d3_owner_check source_association.cpp)
target_compile_features(d3_owner_check PRIVATE cxx_std_20)
target_compile_definitions(d3_owner_check PRIVATE VK_USE_PLATFORM_WIN32_KHR NOMINMAX)
target_compile_options(d3_owner_check PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
target_include_directories(d3_owner_check PRIVATE "${D3_SOURCE}/modules/function/render/vulkan/sinclude")
target_link_libraries(d3_owner_check PRIVATE lux::engine::function::render_vulkan Vulkan::Vulkan)
