cmake_minimum_required(VERSION 3.24)
project(editor_ec1_primitives LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(EC1_MODE "HEADLESS" CACHE STRING "HEADLESS or WINDOW")
if(EC1_MODE STREQUAL "HEADLESS")
    find_package(lux-engine-editor-storage REQUIRED COMPONENTS editor_storage)
    find_package(lux-engine-editor-material-persistence REQUIRED COMPONENTS material_persistence)
    find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence_execution)
    find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
    add_executable(ec1_consumer headless.cpp)
    target_link_libraries(ec1_consumer PRIVATE lux::engine::editor::editor_storage
        lux::engine::editor::material_persistence lux::engine::editor::editor_persistence_execution
        lux::engine::editor::editor_file_publication)
elseif(EC1_MODE STREQUAL "WINDOW")
    find_package(lux-engine-editor-extensions REQUIRED COMPONENTS editor_extensions)
    add_executable(ec1_consumer window.cpp)
    target_link_libraries(ec1_consumer PRIVATE lux::engine::editor::editor_extensions)
else()
    message(FATAL_ERROR "Unknown EC1_MODE")
endif()
foreach(forbidden editor_bootstrap editor_builtin_contributions material_ui flow_ui scene_ui)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "Minimal capability consumer imported ${forbidden}")
    endif()
endforeach()
if(MSVC)
    target_compile_options(ec1_consumer PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
else()
    target_compile_options(ec1_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.ec1.${EC1_MODE} COMMAND ec1_consumer "${CMAKE_CURRENT_BINARY_DIR}/project")
