cmake_minimum_required(VERSION 3.22)
project(lux_editor_editing_consumer LANGUAGES CXX)
find_package(lux-engine-editor-editing REQUIRED COMPONENTS editor_editing)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(business IN ITEMS Text Records)
    add_library(editing_consumer_${business} SHARED ${business}.cpp)
    target_link_libraries(editing_consumer_${business} PRIVATE lux::engine::editor::editor_editing)
    if(MSVC)
        target_compile_options(editing_consumer_${business} PRIVATE /UNDEBUG)
    else()
        target_compile_options(editing_consumer_${business} PRIVATE -UNDEBUG)
    endif()
endforeach()
add_executable(lux_editor_editing_consumer main.cpp)
target_link_libraries(lux_editor_editing_consumer PRIVATE
    lux::engine::editor::editor_editing editing_consumer_Text editing_consumer_Records
)
if(MSVC)
    target_compile_options(lux_editor_editing_consumer PRIVATE /UNDEBUG)
else()
    target_compile_options(lux_editor_editing_consumer PRIVATE -UNDEBUG)
endif()
get_target_property(editing_type lux::engine::editor::editor_editing TYPE)
if(NOT editing_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "ED-1 requires the single shared editing implementation")
endif()
get_target_property(editing_includes lux::engine::editor::editor_editing INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(editing_links lux::engine::editor::editor_editing INTERFACE_LINK_LIBRARIES)
file(WRITE "${CMAKE_BINARY_DIR}/editing-import.txt" "includes=${editing_includes}\nlinks=${editing_links}\n")
option(ED1_WITH_CONTEXT "Also qualify installed Controller/meta" OFF)
if(ED1_WITH_CONTEXT)
    find_package(lux-engine-editor-context REQUIRED COMPONENTS editor_context)
    add_executable(lux_editor_editing_context_consumer Context.cpp)
    target_link_libraries(lux_editor_editing_context_consumer PRIVATE lux::engine::editor::editor_context)
    if(MSVC)
        target_compile_options(lux_editor_editing_context_consumer PRIVATE /UNDEBUG)
    else()
        target_compile_options(lux_editor_editing_context_consumer PRIVATE -UNDEBUG)
    endif()
endif()
