cmake_minimum_required(VERSION 3.24)
project(p11_runtime_sdk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
include("${SDK_PREFIX}/share/lux-engine/plugins/LuxPlugins.cmake")
find_package(lux-engine-project-plugins REQUIRED COMPONENTS project_plugins)
find_package(lux-engine-simulation-ecs REQUIRED COMPONENTS transform)
configure_file(../Identity.cpp.in "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp" @ONLY)
add_library(p11_runtime SHARED ../Runtime.cpp "${CMAKE_CURRENT_BINARY_DIR}/Identity.cpp")
target_link_libraries(p11_runtime PRIVATE lux::engine::simulation::ecs::transform)
add_executable(p11_runtime_consumer main.cpp)
target_link_libraries(p11_runtime_consumer PRIVATE lux::engine::project::project_plugins
    lux::engine::simulation::ecs::transform)
foreach(target p11_runtime p11_runtime_consumer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
get_property(imports DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(target IN LISTS imports)
    if(target MATCHES "lux::engine::editor::")
        message(FATAL_ERROR "Runtime-only consumer imported Editor target ${target}")
    endif()
endforeach()
enable_testing()
add_test(NAME installed.p11.runtime COMMAND p11_runtime_consumer "$<TARGET_FILE:p11_runtime>")
add_test(NAME installed.p12.builtin_plugin_closure COMMAND p11_runtime_consumer --builtins "${SDK_PREFIX}")
