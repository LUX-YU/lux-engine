cmake_minimum_required(VERSION 3.24)
project(object_ownership_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(lux-engine-core REQUIRED COMPONENTS object)
find_package(lux-engine-platform REQUIRED COMPONENTS dynamic_library)

# Reuse maintained behavioral tests; no production source or private include path is imported.
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(file ownership.cpp ownership_plugin.cpp)
    configure_file("${engine_source}/modules/core/object/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
endforeach()
add_library(object_ownership_plugin MODULE "${CMAKE_BINARY_DIR}/ownership_plugin.cpp")
target_link_libraries(object_ownership_plugin PRIVATE lux::engine::core::object)
add_executable(object_ownership "${CMAKE_BINARY_DIR}/ownership.cpp")
target_link_libraries(object_ownership PRIVATE lux::engine::core::object lux::engine::platform::dynamic_library)
if(MSVC)
    target_compile_options(object_ownership PRIVATE /UNDEBUG /permissive- /utf-8)
    target_compile_options(object_ownership_plugin PRIVATE /permissive- /utf-8)
else()
    target_compile_options(object_ownership PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.object.ownership COMMAND object_ownership $<TARGET_FILE:object_ownership_plugin>)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
configure_file("${engine_source}/modules/core/object/test/retirement_contract.py"
    "${CMAKE_BINARY_DIR}/retirement_contract.py" COPYONLY)
add_test(NAME installed.object.retirement_contract COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_BINARY_DIR}/retirement_contract.py" $<TARGET_FILE:object_ownership>)
