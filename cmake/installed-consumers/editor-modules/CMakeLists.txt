cmake_minimum_required(VERSION 3.24)
project(editor_module_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
if(NOT SDK_PREFIX)
    message(FATAL_ERROR "SDK_PREFIX must select the installed SDK")
endif()
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-cxx REQUIRED COMPONENTS reflection_generator)
find_package(lux-engine-editor-extensions REQUIRED COMPONENTS editor_extensions)
find_package(lux-engine-process-execution REQUIRED COMPONENTS process_execution)
include("${SDK_PREFIX}/share/lux-engine/plugins/LuxPlugins.cmake")
include_component_cmake_scripts(lux::engine::core::meta)
include_component_cmake_scripts(lux::engine::editor::editor_extensions)
set(LUX_GENERATE_HEADER_DIR "${CMAKE_BINARY_DIR}/generated")

# Test sources are copied; all production headers, libraries and generation tools come from SDK_PREFIX.
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(file ModuleFixture.hpp ModuleFixture.cpp ModuleExports.cpp modules.cpp)
    configure_file("${engine_source}/editor/application/extensions/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
endforeach()
foreach(file CompositionProbe.hpp composition_plugin.cpp)
    configure_file("${engine_source}/editor/workbench/desktop/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
endforeach()
set(MODULE_NEGATIVE "NONE" CACHE STRING "NONE or SIGNATURE; the invalid declaration must fail generated static_assert")
if(MODULE_NEGATIVE STREQUAL "SIGNATURE")
    file(APPEND "${CMAKE_BINARY_DIR}/ModuleFixture.hpp"
        "\nnamespace module_fixture { LUX_META(luxmodule) int wrongSignature() noexcept; }\n")
    file(APPEND "${CMAKE_BINARY_DIR}/ModuleFixture.cpp"
        "\nnamespace module_fixture { int wrongSignature() noexcept { return 7; } }\n")
elseif(NOT MODULE_NEGATIVE STREQUAL "NONE")
    message(FATAL_ERROR "Unknown MODULE_NEGATIVE")
endif()
foreach(kind static dynamic)
    if(kind STREQUAL "static")
        add_library(module_${kind} STATIC "${CMAKE_BINARY_DIR}/ModuleFixture.cpp" "${CMAKE_BINARY_DIR}/composition_plugin.cpp")
    else()
        add_library(module_${kind} MODULE "${CMAKE_BINARY_DIR}/ModuleFixture.cpp" "${CMAKE_BINARY_DIR}/ModuleExports.cpp"
            "${CMAKE_BINARY_DIR}/composition_plugin.cpp")
    endif()
    target_include_directories(module_${kind} PUBLIC "${CMAKE_BINARY_DIR}")
    target_link_libraries(module_${kind} PUBLIC lux::engine::editor::editor_extensions PRIVATE lux::engine::process::process_execution)
    target_compile_definitions(module_${kind} PRIVATE "MODULE_RUNTIME_ABI=\"${LUX_PLUGIN_SDK_ABI}\"")
endforeach()
engine_declare_editor_module(TARGET module_static HEADER "${CMAKE_BINARY_DIR}/ModuleFixture.hpp"
    LOGICAL_PATH ModuleFixture.hpp SOURCE_FILE "${CMAKE_BINARY_DIR}/ModuleFixture.cpp")
add_executable(module_consumer "${CMAKE_BINARY_DIR}/modules.cpp")
target_link_libraries(module_consumer PRIVATE lux::engine::editor::editor_extensions lux::engine::process::process_execution)
engine_target_add_editor_modules(TARGET module_consumer NAME selectedModules NAMESPACE module_fixture MODULES module_static)
add_executable(empty_consumer empty.cpp)
target_link_libraries(empty_consumer PRIVATE lux::engine::editor::editor_extensions)
engine_target_add_editor_modules(TARGET empty_consumer NAME emptyModules NAMESPACE module_fixture)
foreach(target module_static module_dynamic module_consumer empty_consumer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
enable_testing()
add_test(NAME installed.editor.modules COMMAND module_consumer $<TARGET_FILE:module_dynamic>)
add_test(NAME installed.editor.modules.empty COMMAND empty_consumer)
