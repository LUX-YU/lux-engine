cmake_minimum_required(VERSION 3.25)
project(p10q_sdk_measurements LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-project-ui REQUIRED COMPONENTS project_ui)
find_package(lux-engine-editor-tasks-ui REQUIRED COMPONENTS tasks_ui)
find_package(lux-engine-editor-material-preview REQUIRED COMPONENTS material_preview)
add_executable(measure main.cpp)
add_executable(canvas_ids canvas_ids.cpp)
find_package(lux-engine-editor-widgets REQUIRED COMPONENTS editor_widgets)
target_link_libraries(canvas_ids PRIVATE lux::engine::editor::editor_widgets)
find_package(lux-engine-editor-scene-ui REQUIRED COMPONENTS scene_ui)
find_package(lux-engine-editor-flowforge-interaction REQUIRED COMPONENTS flowforge_interaction)
find_package(lux-engine-scene-transform REQUIRED COMPONENTS scene_transform)
add_executable(candidates candidates.cpp)
target_link_libraries(candidates PRIVATE lux::engine::editor::scene_ui lux::engine::editor::flowforge_interaction
 lux::engine::scene::scene_transform)
add_executable(records records.cpp)
target_link_libraries(records PRIVATE lux::engine::editor::editor_persistence)
if(QUALITY_AFTER)
 find_package(lux-engine-editor-flowforge-compilation REQUIRED COMPONENTS flowforge_compilation)
 add_executable(owners owners.cpp)
 target_link_libraries(owners PRIVATE lux::engine::editor::project_ui lux::engine::editor::tasks_ui
  lux::engine::editor::material_preview lux::engine::editor::flowforge_compilation)
 add_executable(canvas_trace canvas_trace.cpp)
 target_link_libraries(canvas_trace PRIVATE lux::engine::editor::editor_widgets)
 if(MSVC)
  target_compile_options(owners PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
  target_compile_options(canvas_trace PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
 endif()
endif()
if(BASELINE_SOURCE)
 add_executable(hierarchy_alloc hierarchy_alloc.cpp)
 target_include_directories(hierarchy_alloc PRIVATE "${BASELINE_SOURCE}/editor/tools/scene/model/src")
 target_link_libraries(hierarchy_alloc PRIVATE lux::engine::editor::scene_model)
 if(MSVC)
  target_compile_options(hierarchy_alloc PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
 endif()
endif()
if(BASELINE_SOURCE)
 target_include_directories(canvas_ids BEFORE PRIVATE "${BASELINE_SOURCE}/editor/widgets/include")
endif()
target_link_libraries(measure PRIVATE lux::engine::editor::project_ui lux::engine::editor::tasks_ui
    lux::engine::editor::material_preview)
if(QUALITY_AFTER)
 target_compile_definitions(measure PRIVATE QUALITY_AFTER=1)
 target_compile_definitions(canvas_ids PRIVATE QUALITY_AFTER=1)
endif()
if(MSVC)
 target_compile_options(measure PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
 target_compile_options(canvas_ids PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
 target_compile_options(candidates PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
 target_compile_options(records PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
endif()
