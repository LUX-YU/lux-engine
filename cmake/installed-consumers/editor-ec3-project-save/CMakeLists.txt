cmake_minimum_required(VERSION 3.24)
project(editor_ec3_project_save LANGUAGES CXX)
if(NOT LUX_TEST_SUPPORT_DIR)
    get_filename_component(LUX_TEST_SUPPORT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../test-support" ABSOLUTE)
endif()
include_directories("${LUX_TEST_SUPPORT_DIR}")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-storage REQUIRED COMPONENTS editor_storage)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
find_package(lux-engine-editor-scene-persistence REQUIRED COMPONENTS scene_persistence)
find_package(lux-engine-editor-material-persistence REQUIRED COMPONENTS material_persistence)
find_package(lux-engine-editor-flowforge-persistence REQUIRED COMPONENTS flowforge_persistence)
foreach(forbidden editor_bootstrap desktop_shell view_host scene_ui material_ui flowforge_ui)
    if(TARGET lux::engine::editor::${forbidden})
        message(FATAL_ERROR "Project saving imported ${forbidden}")
    endif()
endforeach()
add_executable(editor_ec3_project_save main.cpp)
target_link_libraries(editor_ec3_project_save PRIVATE
    lux::engine::editor::editor_storage lux::engine::editor::editor_file_publication
    lux::engine::editor::scene_persistence lux::engine::editor::material_persistence
    lux::engine::editor::flowforge_persistence)
if(MSVC)
    target_compile_options(editor_ec3_project_save PRIVATE /UNDEBUG /permissive- /utf-8)
else()
    target_compile_options(editor_ec3_project_save PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.ec3.project_save COMMAND editor_ec3_project_save "${CMAKE_BINARY_DIR}/project")
set_tests_properties(installed.ec3.project_save PROPERTIES TIMEOUT 90)
