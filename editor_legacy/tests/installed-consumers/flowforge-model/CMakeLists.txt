cmake_minimum_required(VERSION 3.22)
project(flowforge_model_consumer LANGUAGES CXX)
if(NOT LUX_TEST_SUPPORT_DIR)
    get_filename_component(LUX_TEST_SUPPORT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../test-support" ABSOLUTE)
endif()
include_directories("${LUX_TEST_SUPPORT_DIR}")
set(CMAKE_CXX_STANDARD 20)
find_package(lux-engine-editor-flowforge-model REQUIRED COMPONENTS flowforge_model)
add_executable(flowforge_model_consumer main.cpp)
target_link_libraries(flowforge_model_consumer PRIVATE lux::engine::editor::flowforge_model)
if(MSVC)
    target_compile_options(flowforge_model_consumer PRIVATE /UNDEBUG /Zc:__cplusplus /permissive- /utf-8)
else()
    target_compile_options(flowforge_model_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME flowforge_model_consumer COMMAND flowforge_model_consumer)
