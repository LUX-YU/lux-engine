cmake_minimum_required(VERSION 3.24)
project(services_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CONSUMER_MODE "CORE" CACHE STRING "CORE or TASKS")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
enable_testing()
if(CONSUMER_MODE STREQUAL "CORE")
    find_package(lux-engine-core REQUIRED COMPONENTS services)
    find_package(lux-engine-platform REQUIRED COMPONENTS dynamic_library)
    foreach(file services.cpp ServiceProbe.hpp plugin.cpp plugin_module.cpp)
        configure_file("${engine_source}/modules/core/services/test/${file}" "${CMAKE_BINARY_DIR}/${file}" COPYONLY)
    endforeach()
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/services.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::core::services)
    add_library(services_plugin MODULE "${CMAKE_BINARY_DIR}/plugin_module.cpp")
    target_link_libraries(services_plugin PRIVATE lux::engine::core::services)
    add_executable(services_plugin_consumer "${CMAKE_BINARY_DIR}/plugin.cpp")
    target_link_libraries(services_plugin_consumer PRIVATE
        lux::engine::core::services lux::engine::platform::dynamic_library)
    set(consumers services_consumer services_plugin services_plugin_consumer)
    add_test(NAME installed.services.core COMMAND services_consumer)
    add_test(NAME installed.services.plugin COMMAND services_plugin_consumer $<TARGET_FILE:services_plugin>)
    foreach(forbidden process_execution ui editor_contracts)
        if(TARGET lux::engine::process::${forbidden} OR TARGET lux::engine::editor::${forbidden}
            OR TARGET lux::engine::function::${forbidden})
            message(FATAL_ERROR "Pure services imported ${forbidden}")
        endif()
    endforeach()
elseif(CONSUMER_MODE STREQUAL "TASKS")
    find_package(lux-engine-core REQUIRED COMPONENTS services)
    find_package(lux-engine-process-execution REQUIRED COMPONENTS process_execution)
    configure_file("${engine_source}/engine/context/test/service_tasks.cpp" "${CMAKE_BINARY_DIR}/tasks.cpp" COPYONLY)
    add_executable(services_consumer "${CMAKE_BINARY_DIR}/tasks.cpp")
    target_link_libraries(services_consumer PRIVATE lux::engine::core::services lux::engine::process::process_execution)
    set(consumers services_consumer)
    add_test(NAME installed.services.tasks COMMAND services_consumer)
else()
    message(FATAL_ERROR "Unknown CONSUMER_MODE")
endif()
foreach(consumer IN LISTS consumers)
    if(MSVC)
        target_compile_options(${consumer} PRIVATE /UNDEBUG /permissive- /utf-8 /GR- /we4541)
    else()
        target_compile_options(${consumer} PRIVATE -UNDEBUG -fno-rtti)
    endif()
endforeach()
