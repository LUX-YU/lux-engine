cmake_minimum_required(VERSION 3.24)
project(editor_component_closure LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Component qualification requires RelWithDebInfo")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

option(SCENE_CORE_ONLY "Verify the Scene package without Editor or Render imports" OFF)
if(SCENE_CORE_ONLY)
    find_package(lux-engine-scene REQUIRED COMPONENTS scene_composition)
    if(TARGET lux::engine::function::render_client OR TARGET lux::engine::scene::scene_render)
        message(FATAL_ERROR "Generic Scene imported a Render target")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/scene_core.cpp"
        "#include <lux/engine/scene/SceneInstance.hpp>\nvolatile auto entry = &lux::scene::SceneInstance::worldDescription;\nint main() { return entry == nullptr; }\n")
    add_executable(scene_core_consumer "${CMAKE_CURRENT_BINARY_DIR}/scene_core.cpp")
    target_link_libraries(scene_core_consumer PRIVATE lux::engine::scene::scene_composition)
    target_compile_options(scene_core_consumer PRIVATE /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    enable_testing()
    add_test(NAME closure.scene_core COMMAND scene_core_consumer)
    return()
endif()

function(check_component package component header symbol)
    find_package(${package} REQUIRED COMPONENTS ${component})
    if(TARGET assimp::assimp)
        set_property(TARGET assimp::assimp PROPERTY MAP_IMPORTED_CONFIG_RELWITHDEBINFO Release)
    endif()
    set(source "${CMAKE_CURRENT_BINARY_DIR}/${component}.cpp")
    file(WRITE "${source}" "#include <${header}>\nvolatile auto entry = &${symbol};\nint main() { return entry == nullptr; }\n")
    add_executable(${component}_consumer "${source}")
    target_link_libraries(${component}_consumer PRIVATE lux::engine::editor::${component})
    target_compile_options(${component}_consumer PRIVATE /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    add_test(NAME closure.${component} COMMAND ${component}_consumer)
endfunction()
