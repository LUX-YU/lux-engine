cmake_minimum_required(VERSION 3.24)
project(lux_spatial_consumer LANGUAGES CXX)
if(NOT CMAKE_BUILD_TYPE STREQUAL "RelWithDebInfo")
    message(FATAL_ERROR "Spatial qualification requires RelWithDebInfo")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(lux-cmake-toolset CONFIG REQUIRED)
find_package(lux-engine-function REQUIRED COMPONENTS render_client)
find_package(lux-engine-scene-mesh-query REQUIRED COMPONENTS scene_mesh_query)
if(TARGET lux::engine::function::render_feature_client OR TARGET lux::engine::function::render_features OR
   TARGET lux::engine::function::render_vulkan OR TARGET lux::engine::scene::scene_render)
    message(FATAL_ERROR "Generic consumer imported builtin Features or Renderer")
endif()
include_component_cmake_scripts(lux::engine::core::meta)
include_component_cmake_scripts(lux::engine::function::render_client)
add_library(probe_feature SHARED Probe.cpp)
target_include_directories(probe_feature PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(probe_feature PRIVATE PROBE_LIBRARY)
target_link_libraries(probe_feature PUBLIC lux::engine::function::render_client)
engine_add_comm_ops(NAME external_probe CLIENT_TARGET probe_feature
    INCLUDE_PREFIX probe/generated TARGET_FILES "${CMAKE_CURRENT_SOURCE_DIR}/include/probe/Probe.hpp"
    CLIENT_EXPORT_MACRO PROBE_PUBLIC CLIENT_VISIBILITY_HEADER probe/visibility.hpp
    BACKEND_EXPORT_MACRO PROBE_PUBLIC BACKEND_VISIBILITY_HEADER probe/visibility.hpp)
add_executable(feature_consumer feature.cpp)
target_link_libraries(feature_consumer PRIVATE probe_feature)
add_executable(query_consumer query.cpp)
target_link_libraries(query_consumer PRIVATE lux::engine::scene::scene_mesh_query)
foreach(target probe_feature feature_consumer query_consumer)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG /utf-8 /Zc:__cplusplus /Zc:preprocessor /Zc:externConstexpr)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
enable_testing()
add_test(NAME spatial.external_feature COMMAND feature_consumer)
add_test(NAME spatial.query_without_renderer COMMAND query_consumer)
