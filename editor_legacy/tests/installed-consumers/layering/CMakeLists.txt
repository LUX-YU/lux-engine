cmake_minimum_required(VERSION 3.22)
project(editor_layering_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CONSUMER_MODE TASKS CACHE STRING "Minimal installed layer consumer")
get_filename_component(engine_source "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
if(CONSUMER_MODE STREQUAL "TASKS")
    find_package(lux-engine-editor-tasks REQUIRED COMPONENTS editor_tasks)
    configure_file("${engine_source}/editor/activities/tasks/test/monitor.cpp" "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" COPYONLY)
    set(provider lux::engine::editor::editor_tasks)
elseif(CONSUMER_MODE STREQUAL "SAVE_CORE")
    find_package(lux-engine-editor-persistence REQUIRED COMPONENTS editor_persistence)
    configure_file("${engine_source}/editor/activities/persistence/test/write_coordinator.cpp" "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" COPYONLY)
    set(provider lux::engine::editor::editor_persistence)
elseif(CONSUMER_MODE STREQUAL "PROJECT")
    find_package(lux-engine-editor-project REQUIRED COMPONENTS editor_project)
    configure_file(project.cpp "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" COPYONLY)
    set(provider lux::engine::editor::editor_project)
elseif(CONSUMER_MODE STREQUAL "LAYOUT")
    find_package(lux-engine-editor-layout REQUIRED COMPONENTS layout_model)
    configure_file(layout.cpp "${CMAKE_CURRENT_BINARY_DIR}/main.cpp" COPYONLY)
    set(provider lux::engine::editor::layout_model)
else()
    message(FATAL_ERROR "Unknown layer consumer mode: ${CONSUMER_MODE}")
endif()
get_property(imports DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(import IN LISTS imports)
    if(import MATCHES "(editor_ui|editor_scene|editor_material|editor_flowforge|editor_context|scene_model|material_model|flowforge_model|imgui|Vulkan|render_runtime|scene_composition|flowforge_compiler)")
        message(FATAL_ERROR "Minimal ${CONSUMER_MODE} consumer imported unrelated target: ${import}")
    endif()
    if(NOT CONSUMER_MODE STREQUAL "TASKS" AND import MATCHES "(process_execution|world_loading|asset_loading)")
        message(FATAL_ERROR "Pure ${CONSUMER_MODE} consumer imported Process: ${import}")
    endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/imported-targets.txt" "${imports}\n")
add_executable(layering_consumer "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
target_link_libraries(layering_consumer PRIVATE ${provider})
if(MSVC)
    target_compile_options(layering_consumer PRIVATE /UNDEBUG /utf-8 /permissive- /Zc:__cplusplus)
else()
    target_compile_options(layering_consumer PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME installed.layering.${CONSUMER_MODE} COMMAND layering_consumer)
