cmake_minimum_required(VERSION 3.28)
project(AppConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-framework REQUIRED COMPONENTS lux_editor_app)
add_executable(app_consumer ../../shutdown.cpp)
target_link_libraries(app_consumer PRIVATE lux::engine::editor::lux_editor_app)
if(MSVC)
    target_compile_options(app_consumer PRIVATE /UNDEBUG /utf-8 /GR- /we4541 /permissive-)
else()
    target_compile_options(app_consumer PRIVATE -UNDEBUG -fno-rtti)
endif()
enable_testing()
add_test(NAME sdk.app_only COMMAND app_consumer "${CMAKE_CURRENT_BINARY_DIR}/app.luxproj")
set_tests_properties(sdk.app_only PROPERTIES TIMEOUT 15)
