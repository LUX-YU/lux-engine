cmake_minimum_required(VERSION 3.24)
project(editor_ec3_settings LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(lux-engine-editor-project-tools-ui REQUIRED COMPONENTS project_tools_ui)
find_package(lux-engine-editor-extensions REQUIRED COMPONENTS editor_extensions)
find_package(lux-engine-editor-file-publication REQUIRED COMPONENTS editor_file_publication)
add_executable(editor_ec3_settings ../../../editor/tests/integration/application/settings_navigation.cpp)
target_link_libraries(editor_ec3_settings PRIVATE lux::engine::editor::project_tools_ui
    lux::engine::editor::editor_extensions lux::engine::editor::editor_file_publication)
if(MSVC)
    target_compile_options(editor_ec3_settings PRIVATE /UNDEBUG /utf-8 /permissive-)
else()
    target_compile_options(editor_ec3_settings PRIVATE -UNDEBUG)
endif()
enable_testing()
foreach(mode scope readonly catalog)
    add_test(NAME installed.settings.${mode} COMMAND editor_ec3_settings "${CMAKE_BINARY_DIR}/data" ${mode})
endforeach()
